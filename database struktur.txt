================================================================
  TABEL 1: artikel
  Digunakan di  : halaman Artikel, Home (recent articles)
  PHP backend   : get_articles.php
================================================================

CREATE TABLE artikel (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    judul       VARCHAR(255)  NOT NULL,           -- Judul artikel
    kategori    VARCHAR(100)  NOT NULL,           -- Contoh: "Mental Health", "Fitness"
    ringkasan   TEXT          NOT NULL,           -- Paragraf singkat / preview
    isi         LONGTEXT,                         -- Isi lengkap artikel (opsional)
    gambar_url  VARCHAR(500),                     -- Path / URL gambar thumbnail
    penulis     VARCHAR(100)  NOT NULL,           -- Nama penulis
    tanggal     DATE          NOT NULL,           -- Tanggal publikasi
    status      ENUM('draft','published')
                DEFAULT 'draft',                  -- Status publikasi
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

Contoh isi:
+----+------------------------------------------+---------------+------------------------------+----------+------------------+-------------+------------+-----------+
| id | judul                                    | kategori      | ringkasan                    | penulis  | gambar_url       | tanggal     | status     |
+----+------------------------------------------+---------------+------------------------------+----------+------------------+-------------+------------+-----------+
|  1 | 6 Tips To Protect Your Mental Health...  | Mental Health | It's normal to feel anxiety  | Martin K | images/art1.jpg  | 2022-01-30  | published  |
|  2 | Unsure About Wearing A Face Mask?...    | Health Tips   | Still following shelter...   | John E   | images/art2.jpg  | 2022-01-30  | published  |
+----+------------------------------------------+---------------+------------------------------+----------+------------------+-------------+------------+-----------+

================================================================
  TABEL 2: statistik
  Digunakan di  : halaman Home (kartu statistik)
  PHP backend   : get_statistics.php

CREATE TABLE statistik (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    nama_stat   VARCHAR(255) NOT NULL,            -- Contoh: "Jumlah Anak yang Dibantu"
    nilai_stat  VARCHAR(100) NOT NULL,            -- Contoh: "30", "90%", "Rp 10 Juta"
    deskripsi   TEXT,                             -- Penjelasan singkat statistik
    urutan      INT DEFAULT 0,                    -- Urutan tampil di halaman
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

Contoh isi:
+----+-----------------------------------+-----------+--------------------------------------+--------+
| id | nama_stat                         | nilai_stat| deskripsi                            | urutan |
+----+-----------------------------------+-----------+--------------------------------------+--------+
|  1 | Jumlah Anak yang Dibantu          | 30        | Anak ABK yang mendapat layanan terapi|   1    |
|  2 | Tingkat Keberhasilan Terapi       | 90%       | Persentase anak yang berkembang      |   2    |
|  3 | Program & Acara Diselenggarakan   | 75        | Total kegiatan yang dilaksanakan     |   3    |
|  4 | Jumlah Donasi yang Diterima       | Rp 10 Juta| Total donasi terkumpul               |   4    |
+----+-----------------------------------+-----------+--------------------------------------+--------+

  TABEL 4: pendaftaran_siswa

CREATE TABLE pendaftaran_siswa (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    nama_siswa      VARCHAR(150) NOT NULL,        -- Nama lengkap siswa ABK
    tempat_lahir    VARCHAR(100),                 -- Tempat lahir
    tanggal_lahir   DATE,                         -- Tanggal lahir
    alamat          TEXT NOT NULL,                -- Alamat tinggal
    nama_orang_tua  VARCHAR(150) NOT NULL,        -- Nama ayah/ibu/wali
    no_telepon      VARCHAR(20)  NOT NULL,        -- Nomor HP orang tua
    status          ENUM('baru','diproses','diterima','ditolak')
                    DEFAULT 'baru',               -- Status proses pendaftaran
    catatan         TEXT,                         -- Catatan admin
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

================================================================
  TABEL 7: donasi
  Digunakan di  : halaman Home (seksi Donasi)
  Bukti dikirim : via Google Form (forms.gle/ZuB8JpHTB56GAR8H9)
================================================================

CREATE TABLE donasi (
    id              INT AUTO_INCREMENT PRIMARY KEY,
    nama_donatur    VARCHAR(150),                 -- Nama donatur (boleh anonim)
    jumlah          DECIMAL(15,2) NOT NULL,       -- Jumlah donasi dalam Rupiah
    metode          VARCHAR(50),                  -- Contoh: OVO, GoPay, Dana, ShopeePay
    bukti_transfer  VARCHAR(500),                 -- Path file bukti transfer (opsional)
    pesan           TEXT,                         -- Pesan dari donatur
    status          ENUM('menunggu','terverifikasi','ditolak')
                    DEFAULT 'menunggu',
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Contoh isi:
+----+---------------+----------+---------+---------+-----------+
| id | nama_donatur  | jumlah   | metode  | pesan   | status    |
+----+---------------+----------+---------+---------+-----------+
|  1 | Anonim        | 50000.00 | GoPay   | Semoga  | terverf.. |
|  2 | Sari Dewi     | 200000.00| OVO     | Sukses  | terverf.. |
+----+---------------+----------+---------+---------+-----------+

================================================================
  TABEL 8: admin_users
  Digunakan di  : halaman Admin Dashboard
================================================================

CREATE TABLE admin_users (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    username    VARCHAR(100) NOT NULL UNIQUE,     -- Username login
    password    VARCHAR(255) NOT NULL,            -- Hash bcrypt
    nama        VARCHAR(150),                     -- Nama lengkap admin
    email       VARCHAR(150) NOT NULL UNIQUE,
    role        ENUM('superadmin','admin','editor')
                DEFAULT 'editor',
    last_login  TIMESTAMP NULL,
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Contoh isi:
+----+----------+-------------+--------------------+------------+
| id | username | nama        | email              | role       |
+----+----------+-------------+--------------------+------------+
|  1 | admin    | Super Admin | admin@act.ac.id    | superadmin |
|  2 | editor1  | Tim Editor  | editor@act.ac.id   | editor     |
+----+----------+-------------+--------------------+------------+
